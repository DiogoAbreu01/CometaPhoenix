# Jogo de corrida 2D + oficina 3D (OpenGL 3.3 / SDL2) - Windows, Linux, macOS.
option(COMETA_GAME "Compilar jogo nativo OpenGL/SDL2" ON)
if(COMETA_GAME)
    # O pacote SDL2 do vcpkg fornece SDL2Config.cmake/SDL2::SDL2,
    # enquanto algumas distros Linux disponibilizam somente o runtime.
    find_package(SDL2 CONFIG QUIET)
    if(TARGET SDL2::SDL2)
        set(COMETA_SDL2_TARGET SDL2::SDL2)
    elseif(TARGET SDL2::SDL2-static)
        set(COMETA_SDL2_TARGET SDL2::SDL2-static)
    else()
        if(NOT WIN32)
            find_library(COMETA_SDL2_LIBRARY NAMES SDL2 SDL2-2.0 libSDL2-2.0.so.0)
            if(COMETA_SDL2_LIBRARY)
                set(COMETA_SDL2_TARGET "${COMETA_SDL2_LIBRARY}")
            endif()
        endif()
    endif()

    if(COMETA_SDL2_TARGET)
        add_executable(cometa_game game/GameApp.cpp)
        target_link_libraries(cometa_game PRIVATE cometa_game_logic ${COMETA_SDL2_TARGET})
        target_compile_definitions(cometa_game PRIVATE COMETA_GAME_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/game/shaders")
        add_executable(cometa_engine_audio_tests tests/test_engine_audio.cpp)
        target_link_libraries(cometa_engine_audio_tests PRIVATE cometa_game_logic ${COMETA_SDL2_TARGET})
        target_compile_definitions(cometa_engine_audio_tests PRIVATE
            $<TARGET_PROPERTY:cometa_game,COMPILE_DEFINITIONS>)
        add_test(NAME cometa_engine_audio COMMAND cometa_engine_audio_tests)
        add_executable(cometa_impact_audio_tests tests/test_impact_audio.cpp)
        add_test(NAME cometa_impact_audio COMMAND cometa_impact_audio_tests)
        if(WIN32 AND TARGET SDL2::SDL2)
            add_custom_command(TARGET cometa_engine_audio_tests POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL2::SDL2>"
                    "$<TARGET_FILE_DIR:cometa_engine_audio_tests>")
        endif()
        if(MSVC)
            target_compile_options(cometa_game PRIVATE /W4 /O2 /EHsc)
            target_compile_definitions(cometa_game PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(cometa_game PRIVATE -O2 -Wall -Wextra -Wpedantic)
        endif()
        install(TARGETS cometa_game RUNTIME DESTINATION .)
        install(DIRECTORY game/shaders/ DESTINATION game/shaders)
        install(DIRECTORY game/assets/ DESTINATION game/assets FILES_MATCHING PATTERN "*.rgb" PATTERN "*.r8" PATTERN "*.jpg" PATTERN "*.rgba" PATTERN "*.mesh")
        if(WIN32 AND TARGET SDL2::SDL2)
            # Empacota a DLL do vcpkg no mesmo diretorio do executavel.
            install(FILES "$<TARGET_FILE:SDL2::SDL2>" DESTINATION .)
        endif()
        message(STATUS "Cometa Phoenix jogo: SDL2 via ${COMETA_SDL2_TARGET}")
    else()
        message(FATAL_ERROR "SDL2 ausente. Instale libsdl2-dev no Linux ou SDL2 via vcpkg no Windows; use -DCOMETA_GAME=OFF apenas para motor CLI.")
    endif()
endif()
